# (/opt/brainsyncos/venv, package brainsync-runtime), one distribution name per
# line.

# For each name listed here, scripts/venv.sh needs exactly one wheel of it in
# build/wheels/, as scripts/build-wheels.sh (scripts/build.sh wheels) leaves

# there that are not listed are refused. They are installed with --no-deps
# after the locked wheels, so their own dependencies must already be in
# brainsync.lock.txt. Their sha256 is recorded in
# /usr/share/doc/brainsync-runtime/wheels.txt.
